* /users                                                               :verb:
template http://localhost:3001

** List users                                                      :success:
get /users

Returns user objects.  The primary key ~id~ is always included; use
~?fields~ to request additional columns:

#+begin_src text
  GET /users
  # → [{"id": "…"}, …]

  GET /users?fields=name
  # → [{"id": "…", "name": "…"}, …]

  GET /users?fields=name,email
  # → [{"id": "…", "name": "…", "email": "…"}, …]
#+end_src

** Get missing user                                                  :error:
# * Returns an HTTP 404 when the user doesn't match
#
#   Example,
#   {
#     "error": "User 00000000-aaaa-bbbb-bbbb-000000000000 not found"
#   }

get /users/00000000-aaaa-bbbb-bbbb-000000000000

** Create user                                               :success:error:
:PROPERTIES:
:Verb-Store: new-user
:END:

# * Returns an HTTP 201 when a new valid user is created
#
# * Returns an HTTP 409 when the user's email is already registered
#
#   Example,
#   {
#     "error": "A user with this email already exists"
#   }
post /users
Content-Type: application/json

#+begin_src json
  {
    "name": "Alice",
    "email": "alice@example.com"
  }
#+end_src


** Get existing user                                               :success:
get /users/{{(verb-json-get (oref (verb-stored-response "new-user") body) "id")}}

** Update user name                                             :success:error:
# Update a user's name with a =PATCH= request.  Only the ~name~ field is
# mutable; ~id~ and ~created_at~ are immutable at the database level
# (enforced via a PostgreSQL trigger).

*** Update user name                                        :success:error:
patch /users/{{(verb-json-get (oref (verb-stored-response "new-user") body) "id")}}
Content-Type: application/json
# This gets an empty response 204

#+begin_src json
  {
    "name": "Alice Updated"
  }
#+end_src

*** Update user name and get representation back (RFC 7240) :success:error:
# This gets a (internal/private filtered) User representation back (200)

patch /users/{{(verb-json-get (oref (verb-stored-response "new-user") body) "id")}}
Content-Type: application/json
Prefer: return=representation

#+begin_src json
{
